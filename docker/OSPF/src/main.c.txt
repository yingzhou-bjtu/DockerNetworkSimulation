#define TABLE_SIZE 1000

#define V 1000  // 图的顶点数量（根据实际情况调整）
#define MAX_QUEUE_SIZE V  // 队列的最大大小

#define MAX_COMMANDS 100
#define COMMAND_SIZE 256
// 定义哈希表节点结构
typedef struct HashNode {
    char *key;
    char **values;
    int valuesSize;
    int valuesCapacity;
    struct HashNode *next;
} HashNode;

// 定义哈希表结构
typedef struct HashTable {
    HashNode *nodes[TABLE_SIZE];
} HashTable;

// 哈希函数
unsigned int hashFunction(const char *str) {
    unsigned int hash = 0;
    for (int i = 0; str[i] != '\0'; i++) {
        hash = (hash * 31 + str[i]) % TABLE_SIZE;
    }
    return hash;
}

// 创建哈希表
HashTable *createHashTable() {
    HashTable *table = (HashTable *)malloc(sizeof(HashTable));
    if (!table) {
        perror("Failed to create hash table");
        exit(EXIT_FAILURE);
    }
    for (int i = 0; i < TABLE_SIZE; i++) {
        table->nodes[i] = NULL;
    }
    return table;
}

// 插入key-value对
void insert(HashTable *table, const char *key, const char *value) {
    if (!table) return;

    unsigned int index = hashFunction(key);
    HashNode *node = table->nodes[index];

    while (node) {
        if (strcmp(node->key, key) == 0) {
            for (int i = 0; i < node->valuesSize; i++) {
                if (strcmp(node->values[i], value) == 0) {
                    return;
                }
            }
            node->valuesSize++;
            node->valuesCapacity = node->valuesSize < node->valuesCapacity ? node->valuesCapacity : node->valuesSize * 2;
            node->values = (char **)realloc(node->values, sizeof(char *) * node->valuesCapacity);
            node->values[node->valuesSize - 1] = strdup(value);
            return;
        }
        node = node->next;
    }

    node = (HashNode *)malloc(sizeof(HashNode));
    node->key = strdup(key);
    node->valuesSize = 1;
    node->valuesCapacity = 1;
    node->values = (char **)malloc(sizeof(char *));
    node->values[0] = strdup(value);
    node->next = table->nodes[index];
    table->nodes[index] = node;
}

// 查找key并返回value数组
char **search(HashTable *table, const char *key, int *size) {
    if (!table || !key || !size) return NULL;

    unsigned int index = hashFunction(key);
    HashNode *node = table->nodes[index];

    while (node) {
        if (strcmp(node->key, key) == 0) {
            *size = node->valuesSize;
            return node->values;
        }
        node = node->next;
    }

    *size = 0;
    return NULL;
}
// 显示哈希表中的所有键值对
void displayHashTable(HashTable *table) {
    if (!table) return;

    for (int i = 0; i < TABLE_SIZE; i++) {
        HashNode *node = table->nodes[i];
        while (node) {
            printf("K: %s, V: ", node->key);
            for (int j = 0; j < node->valuesSize; j++) {
                printf("%s ", node->values[j]);
            }
            printf("\n");
            node = node->next;
        }
    }
}

// 收集所有设备
int collectAllDevices(HashTable *table, char ***allDevices) {
    int totalDevices = 0;
    *allDevices = NULL;

    for (int i = 0; i < TABLE_SIZE; i++) {
        HashNode *node = table->nodes[i];
        while (node) {
            for (int j = 0; j < node->valuesSize; j++) {
                int found = 0;
                for (int k = 0; k < totalDevices; k++) {
                    if (strcmp((*allDevices)[k], node->values[j]) == 0) {
                        found = 1;
                        break;
                    }
                }
                if (!found) {
                    totalDevices++;
                    *allDevices = (char **)realloc(*allDevices, totalDevices * sizeof(char *));
                    (*allDevices)[totalDevices - 1] = strdup(node->values[j]);
                }
            }
            node = node->next;
        }
    }

    return totalDevices;
}

// 建立邻接矩阵
void buildAdjacencyMatrix(HashTable *table, char **allDevices, int totalDevices, int **adjMatrix) {
    for (int i = 0; i < totalDevices; i++) {
        for (int j = 0; j < totalDevices; j++) {
            adjMatrix[i][j] = 0;
        }
    }

    for (int i = 0; i < TABLE_SIZE; i++) {
        HashNode *node = table->nodes[i];
        while (node) {
            for (int j = 0; j < node->valuesSize; j++) {
                for (int k = 0; k < totalDevices; k++) {
                    if (strcmp(node->values[j], allDevices[k]) == 0) {
                        for (int l = 0; l < node->valuesSize; l++) {
                            for (int m = 0; m < totalDevices; m++) {
                                if (strcmp(node->values[l], allDevices[m]) == 0 && k != m) {
                                    adjMatrix[k][m] = 1;
                                    adjMatrix[m][k] = 1;
                                }
                            }
                        }
                    }
                }
            }
            node = node->next;
        }
    }
}

typedef struct {
    int id[V];
    int next_hop[V];
} BFSResult;

typedef struct {
    int data[MAX_QUEUE_SIZE];
    int front, rear;
} Queue;

// 初始化队列
void initQueue(Queue* q) {
    q->front = 0;
    q->rear = -1;
}

// 入队
void enqueue(Queue* q, int value) {
    if (q->rear < MAX_QUEUE_SIZE - 1) {
        q->data[++(q->rear)] = value;
    }
}

// 出队
int dequeue(Queue* q) {
    if (q->front <= q->rear) {
        return q->data[(q->front)++];
    }
    return -1; // 队列为空时返回 -1
}

// 检查队列是否为空
bool isQueueEmpty(Queue* q) {
    return q->front > q->rear;
}

// BFS 计算函数
BFSResult bfs(int **graph, int totalDevices, int start) {
    BFSResult result;
    int dist[V]; // 存储从起始节点到其他节点的距离
    bool visited[V]; // 访问标记
    int prev[V]; // 存储每个节点的前驱节点
    Queue q;

    initQueue(&q);

    // 初始化
    for (int i = 0; i < V; i++) {
        dist[i] = -1; // 初始化为 -1，表示不可达
        visited[i] = false;
        prev[i] = -1;
        result.next_hop[i] = -1; // 初始化下一跳为 -1
    }
    dist[start] = 0;
    visited[start] = true;
    enqueue(&q, start);

    // BFS
    while (!isQueueEmpty(&q)) {
        int u = dequeue(&q);

        for (int v = 0; v < totalDevices; v++) {
            if (graph[u][v] == 1 && !visited[v]) {
                visited[v] = true;
                dist[v] = dist[u] + 1;
                prev[v] = u;
                enqueue(&q, v);
            }
        }
    }

    // 处理下一跳
    for (int i = 0; i < totalDevices; i++) {
        result.id[i] = i;

        if (dist[i] != -1) {
            // 从目标节点追溯到起始节点
            int u = i;
            while (prev[u] != start && prev[u] != -1) {
                u = prev[u];
            }
            result.next_hop[i] = (prev[u] == start) ? u : -1;
        } else {
            result.next_hop[i] = -1;
        }
    }

    return result;
}

// 假设的命令存储结构
typedef struct {
    char command[COMMAND_SIZE];
    int executed;
} executed_command;

executed_command command_history[MAX_COMMANDS];
int command_count = 0;

// 检查命令是否已执行
int command_already_executed(const char *command) {
    for (int i = 0; i < command_count; i++) {
        if (strcmp(command_history[i].command, command) == 0) {
            return 1; // 命令已执行
        }
    }
    return 0; // 命令未执行
}

// 添加命令到历史记录
void add_command_to_history(const char *command) {
    if (command_count < MAX_COMMANDS) {
        strcpy(command_history[command_count].command, command);
        command_history[command_count].executed = 1;
        command_count++;
    }
}

// 执行命令，如果它尚未执行
void execute_command_if_not_already(const char *command) {
    if (!command_already_executed(command)) {
        int sta = system(command);
        if (sta == -1) {
            printf("Error: Command execution failed.\n");
        } else {
            add_command_to_history(command); // 命令执行成功，添加到历史记录
        }
    }
}

// 释放单个哈希节点
void freeHashNode(HashNode *node) {
    if (!node) return;

    // 释放键的内存
    free(node->key);

    // 释放所有值的内存
    for (int i = 0; i < node->valuesSize; i++) {
        free(node->values[i]);
    }
    // 释放值数组的内存
    free(node->values);

    // 释放节点本身
    free(node);
}

// 释放整个哈希表
void freeHashTable(HashTable *table) {
    if (!table) return;

    // 遍历哈希表中的每一个槽
    for (int i = 0; i < TABLE_SIZE; i++) {
        HashNode *node = table->nodes[i];

        // 逐个释放链表中的节点
        while (node) {
            HashNode *nextNode = node->next;
            freeHashNode(node);
            node = nextNode;
        }
    }

    // 释放哈希表结构体的内存
    free(table);
}




// DFS 遍历图的连通性
void dfs(int **graph, int node, bool *visited, int totalDevices) {
    visited[node] = true;
    for (int i = 0; i < totalDevices; i++) {
        if (graph[node][i] == 1 && !visited[i]) {
            dfs(graph, i, visited, totalDevices);
        }
    }
}

// 检查图是否连通
bool isConnected(int **graph, int totalDevices) {
    if (totalDevices == 0) return true;  // 如果图没有顶点，认为是连通的

    bool visited[V] = {false};

    // 从第一个顶点开始进行DFS
    dfs(graph, 0, visited, totalDevices);

    // 检查是否所有顶点都被访问过
    for (int i = 0; i < totalDevices; i++) {
        if (!visited[i]) {
            return false;  // 有未访问的顶点，图不连通
        }
    }

    return true;  // 所有顶点都被访问，图是连通的
}
